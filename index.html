How it works

This All-in-One Google Apps Script fetches live Hindi news for all 8 cards (Atrauli Local + Aligarh + UP + National + Amar Ujala, Bhaskar, Hindustan, Jagran) in a single request.

function fetchFeedItems(query, maxCount) {
  try {
    var url = 'https://news.google.com/rss/search?q=' + encodeURIComponent(query) + '&hl=hi&gl=IN&ceid=IN:hi&scoring=n';
    var xml = UrlFetchApp.fetch(url, { muteHttpExceptions: true }).getContentText();
    var doc = XmlService.parse(xml);
    var items = doc.getRootElement().getChild('channel').getChildren('item');
    var out = [];
    for (var i = 0; i < Math.min(items.length, maxCount || 8); i++) {
      var srcEl = items[i].getChild('source');
      out.push({
        title: items[i].getChildText('title'),
        link: items[i].getChildText('link'),
        pubDate: items[i].getChildText('pubDate'),
        source: srcEl ? srcEl.getText() : ''
      });
    }
    return out;
  } catch (err) {
    return [];
  }
}

function doGet(e) {
  var feeds = {
    atrauli: fetchFeedItems('अतरौली', 10),
    aligarh: fetchFeedItems('अलीगढ़ समाचार', 8),
    up: fetchFeedItems('उत्तर प्रदेश बड़ी खबर', 8),
    national: fetchFeedItems('PIB India प्रेस विज्ञप्ति भारत सरकार', 8),
    paper_amarujala: fetchFeedItems('site:amarujala.com अतरौली', 8),
    paper_bhaskar: fetchFeedItems('site:bhaskar.com अतरौली', 8),
    paper_hindustan: fetchFeedItems('site:livehindustan.com अतरौली', 8),
    paper_jagran: fetchFeedItems('site:jagran.com अतरौली', 8)
  };
  if (feeds.atrauli.length === 0) {
    feeds.atrauli = fetchFeedItems('Atrauli Aligarh news', 8);
  }
  var jsonStr = JSON.stringify({ status: 'ok', feeds: feeds });
  var cb = e && e.parameter && e.parameter.callback;
  if (cb) {
    var safeCb = String(cb).replace(/[^a-zA-Z0-9_$.]/g, '');
    return ContentService.createTextOutput(safeCb + '(' + jsonStr + ');')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(jsonStr)
    .setMimeType(ContentService.MimeType.JSON);
}
